### Додавання TTL у Guard

**PATH:** `vfoundation/apps/reference/domains/execution_position/exposure_guard.py`

```python
import time
from decimal import Decimal
# ...

class ExposureGuard:
    def __init__(self, cfg, logger):
        # ...
        ex = (self.cfg.get("execution") or {}).get("exposure") or {}
        self.ttl_sec = int(ex.get("pending_ttl_sec", 300))  # 5 хв дефолт
        # state.reservations: key -> {"notional": Decimal, "ts": int, "reduce_only": bool}

    def reserve(self, key: str, notional_usd: Decimal, *, reduce_only: bool = False) -> None:
        if self.exclude_reduce_only and reduce_only:
            return
        now = int(time.time())
        self.state.reservations[key] = {"notional": notional_usd, "ts": now, "reduce_only": reduce_only}
        if self.count_pending:
            self.state.pending_open_usd += notional_usd

    def release(self, key: str) -> None:
        rec = self.state.reservations.pop(key, None)
        if rec and self.count_pending:
            self.state.pending_open_usd = max(Decimal("0"), self.state.pending_open_usd - rec["notional"])

    def cleanup_expired(self) -> list[str]:
        if self.ttl_sec <= 0:
            return []
        now = int(time.time())
        expired = [k for k, v in self.state.reservations.items() if now - v["ts"] >= self.ttl_sec]
        for k in expired:
            self.release(k)
        return expired
```

```python
from time import time
from dataclasses import dataclass, field

@dataclass
class ExposureState:
    equity_free_usdt: Decimal = Decimal("0")
    open_positions_usd: Decimal = Decimal("0")
    pending_open_usd: Decimal = Decimal("0")
    reservations: Dict[str, Decimal] = field(default_factory=dict)
    reservations_ts: Dict[str, float] = field(default_factory=dict)  # key -> epoch_s

class ExposureGuard:
    def __init__(self, cfg: dict, logger):
        ...
        ex = (self.cfg.get("execution") or {}).get("exposure") or {}
        self.ttl_sec = int(ex.get("pending_reservation_ttl_sec", 90))  # дефолт 90� 

    def reserve(self, key: str, notional_usd: Decimal, *, reduce_only: bool = False) -> None:
        if self.exclude_reduce_only and reduce_only:
            return
        self.state.reservations[key] = notional_usd
        self.state.reservations_ts[key] = time()
        if self.count_pending:
            self.state.pending_open_usd += notional_usd

    def release(self, key: str) -> None:
        val = self.state.reservations.pop(key, None)
        self.state.reservations_ts.pop(key, None)
        if val is not None and self.count_pending:
            self.state.pending_open_usd = max(Decimal("0"), self.state.pending_open_usd - val)

    def expire_stale(self) -> list[str]:
        """Звільняє про� трочені резерви; повертає � пи� ок ключів, які знято."""
        now = time()
        expired = []
        for k, ts in list(self.state.reservations_ts.items()):
            if now - ts >= self.ttl_sec:
                self.release(k)
                expired.append(k)
        return expired
```

```python
from typing import Any

try:
    from prometheus_client import Counter, Gauge, generate_latest, CONTENT_TYPE_LATEST
except Exception:  # fail-closed: заглушки якщо либ немає
    class _D:
        def __getattr__(self, _): return self
        def labels(self, *_, **__): return self
        def set(self, *_): pass
        def inc(self, *_): pass
    Gauge = Counter = _D  # type: ignore
    def generate_latest(): return b""
    CONTENT_TYPE_LATEST = "text/plain; version=0.0.4"

        from decimal import Decimal as D
        return float(D(str(x)))
    except (InvalidOperation, ValueError, TypeError):
        return 0.0
```

```python
import time
from collections import defaultdict, deque
from decimal import Decimal, InvalidOperation
from vfoundation.core.protocol import Message
from vfoundation.apps.reference.telemetry.metrics import inc_decision_rl

def _d(x):
    try:
        return Decimal(str(x))
    except (InvalidOperation, ValueError, TypeError):
        return Decimal("0")

class DecisionMaking:
    def __init__(self, fsm, cfg, logger):
        self.fsm = fsm
        self.cfg = cfg or {}
        self.logger = logger
        qos = ((self.cfg.get("decision") or {}).get("qos") or {})
        self._cooldown_after_exposure_block = int(qos.get("exposure_block_cooldown_sec", 10))
        self._symbol_cooldown_sec = int(qos.get("symbol_cooldown_sec", 3))
        self._max_ipm = int(qos.get("max_intents_per_minute_per_symbol", 6))
        # QoS state
        self._last_block_ts = defaultdict(lambda: 0.0)     # symbol -> ts (block by exposure)
        self._last_intent_ts = defaultdict(lambda: 0.0)    # symbol -> ts (last propose)
        self._minute_window = defaultdict(lambda: deque(maxlen=60))  # symbol -> deque[timestamps]

        # portfolio/equity cache
        self._equity_usdt = Decimal("0")
        self._positions_cache = []

    def on_portfolio(self, msg: Message):
        pld = msg.pld or {}
        eq = _d(pld.get("equity_free_usdt"))
        if eq > 0 or getattr(self, "_equity_usdt", Decimal("0")) <= 0:
            self._equity_usdt = eq
        self._positions_cache = pld.get("positions") or []
        self.logger.info("   Equity (free): %s, Positions: %d", str(self._equity_usdt), len(self._positions_cache))

    def on_err_open(self, msg: Message):
        # викликати з bridge, коли приходить ERR:OPEN
        pld = msg.pld or {}
        if pld.get("reason") == "PORTFOLIO_EXPOSURE_LIMIT":
            sym = (pld.get("symbol") or pld.get("instrument") or "").upper()
            if sym:
                self._last_block_ts[sym] = time.time()

    def _qos_allow(self, symbol: str) -> tuple[bool, str]:
        now = time.time()
        # 1) cooldown пі� ля exposure block
        block_ts = self._last_block_ts[symbol]
        if block_ts and (now - block_ts) < self._cooldown_after_exposure_block:
            return False, "qos_exposure_block_cooldown"

        # 2) symbol cooldown
        last_ts = self._last_intent_ts[symbol]
        if last_ts and (now - last_ts) < self._symbol_cooldown_sec:
            return False, "qos_symbol_cooldown"

        # 3) intents per minute
        dq = self._minute_window[symbol]
        # очи� тити � тарші за 60� 
        while dq and (now - dq[0]) > 60.0:
            dq.popleft()
        if len(dq) >= self._max_ipm:
            return False, "qos_rate_limited"

        return True, "ok"

    async def maybe_propose(self, symbol: str, build_intent_cb):
        allowed, reason = self._qos_allow(symbol)
        if not allowed:
            self.logger.warning("[%s] Decision QoS: %s", symbol, reason)
            inc_decision_rl(symbol)
            await self.fsm.emit(Message(
                op="EVT", verb="DECISION_RATE_LIMITED", intent="OBSERVATION",
                src="decision_making", dst="any", rid=None,
                pld={"symbol": symbol, "reason": reason}, why="decision_qos"
            ))
            return

        # згенерувати intent
        intent = build_intent_cb()
        if not intent:
            return
        self._last_intent_ts[symbol] = time.time()
        self._minute_window[symbol].append(self._last_intent_ts[symbol])
        await self.fsm.emit(Message(
            op="EVT", verb="TRADE_INTENT_PROPOSED", intent="INTENT",
            src="decision_making", dst="execution_position", rid=intent["rid"],
            pld=intent, why="intent_ready"
        ))
```

```python
# canonical reasons
NRR = {
    "IMMEDIATE_TRIGGER": "NRR_IMMEDIATE_TRIGGER",
    "MIN_NOTIONAL": "NRR_MIN_NOTIONAL",
    "MIN_QTY": "NRR_MIN_QTY",
    "PRICE_TICK": "NRR_PRICE_TICK",
    "RATE_LIMIT": "NRR_RATE_LIMIT",
    "SIGNATURE_INVALID": "NRR_INVALID_SIGNATURE",
    "REDUCE_ONLY_RULE": "NRR_REDUCE_ONLY_RULE",
    "POSITION_SIDE_CONFLICT": "NRR_POSITION_SIDE_CONFLICT",
    "UNKNOWN": "NRR_UNKNOWN",
}

_PATTERNS: Dict[str, re.Pattern] = {
    NRR["IMMEDIATE_TRIGGER"]: re.compile(r"would\s+immediately\s+trigger", re.I),
    NRR["MIN_NOTIONAL"]: re.compile(r"min[_\s]?notional", re.I),
    NRR["MIN_QTY"]: re.compile(r"min[_\s]?qty|LOT_SIZE", re.I),
    NRR["PRICE_TICK"]: re.compile(r"tick[_\s]?size|min[_\s]?price|max[_\s]?price", re.I),
    NRR["RATE_LIMIT"]: re.compile(r"rate\s*limit|too many requests|429", re.I),
    NRR["SIGNATURE_INVALID"]: re.compile(r"-?1022|signature\s+is\s+not\s+valid", re.I),
    NRR["REDUCE_ONLY_RULE"]: re.compile(r"reduceOnly.*(rejected|not.*allowed)", re.I),
    NRR["POSITION_SIDE_CONFLICT"]: re.compile(r"positionSide", re.I),
}

    if not msg: return (NRR["UNKNOWN"], "empty")
    for code, pat in _PATTERNS.items():
        if pat.search(msg):
            return (code, msg)
    return (NRR["UNKNOWN"], msg)
```

```python
from __future__ import annotations
from dataclasses import dataclass, field
from decimal import Decimal
from time import time
from typing import Optional, Dict

@dataclass
class OrderRef:
    rid: str
    idempotent_key: Optional[str]
    clientOrderId: Optional[str] = None
    exchangeOrderId: Optional[str] = None
    symbol: Optional[str] = None
    side: Optional[str] = None
    order_type: Optional[str] = None
    created_ts: float = field(default_factory=time)
    terminal: bool = False

class OrderIndex:
    def __init__(self, ttl_sec: int = 3600):
        self.ttl = ttl_sec
        self._by_rid: Dict[str, OrderRef] = {}
        self._by_client: Dict[str, OrderRef] = {}
        self._by_exchange: Dict[str, OrderRef] = {}

    def upsert_from_open(self, *, rid: str, idempotent_key: str, clientOrderId: Optional[str], symbol: str, side: str, order_type: str) -> OrderRef:
        ref = self._by_rid.get(rid) or OrderRef(rid=rid, idempotent_key=idempotent_key)
        ref.clientOrderId = clientOrderId or ref.clientOrderId
        ref.symbol, ref.side, ref.order_type = symbol, side, order_type
        self._by_rid[rid] = ref
        if ref.clientOrderId:
            self._by_client[ref.clientOrderId] = ref
        return ref

    def attach_exchange_id(self, *, clientOrderId: Optional[str], exchangeOrderId: Optional[str]) -> Optional[OrderRef]:
        ref = (clientOrderId and self._by_client.get(clientOrderId)) or None
        if ref and exchangeOrderId:
            ref.exchangeOrderId = exchangeOrderId
            self._by_exchange[exchangeOrderId] = ref
        return ref

    def get(self, *, rid: str=None, clientOrderId: str=None, exchangeOrderId: str=None) -> Optional[OrderRef]:
        if rid: return self._by_rid.get(rid)
        if clientOrderId: return self._by_client.get(clientOrderId)
        if exchangeOrderId: return self._by_exchange.get(exchangeOrderId)
        return None

    def mark_terminal(self, ref: OrderRef) -> None:
        ref.terminal = True

    def expire(self) -> int:
        now = time(); removed = 0
        def _kill(ref: OrderRef):
            nonlocal removed
            self._by_rid.pop(ref.rid, None); removed += 1
            if ref.clientOrderId: self._by_client.pop(ref.clientOrderId, None)
            if ref.exchangeOrderId: self._by_exchange.pop(ref.exchangeOrderId, None)

        for ref in list(self._by_rid.values()):
            if ref.terminal or (now - ref.created_ts) > self.ttl:
                _kill(ref)
        return removed
```

```python
from __future__ import annotations
from dataclasses import dataclass
from datetime import datetime, timezone
from decimal import Decimal, InvalidOperation
from typing import Optional, Dict, Any, Tuple

def _d(x) -> Decimal:
    from decimal import Decimal as D
    try: return D(str(x))
    except (InvalidOperation, ValueError, TypeError): return D("0")

def _now_utc() -> datetime:
    return datetime.now(timezone.utc)

@dataclass
class DailyConfig:
    max_realized_loss_usd: Decimal
    max_drawdown_pct: Decimal
    reset_h: int
    reset_m: int

class DailyRiskState:
    """
    Тримає денно-агреговані метрики:
      - equity_open_usd (snapshot на добовому ре� еті),
      - equity_now_usd (оновлюєть� я від портфеля),
      - realized_pnl_usd (накопичуємо з FILLED/закриттів).
    """
    def __init__(self, cfg: dict, logger=None):
        self.log = logger
        rcfg = ((cfg.get("risk") or {}).get("daily") or {})
        self.cfg = DailyConfig(
            max_realized_loss_usd=_d(rcfg.get("max_realized_loss_usd", 250)),
            max_drawdown_pct=_d(rcfg.get("max_drawdown_pct", 8)),
            reset_h=int(str(rcfg.get("reset_time_utc", "00:00")).split(":")[0]),
            reset_m=int(str(rcfg.get("reset_time_utc", "00:00")).split(":")[1]),
        )
        self._equity_open = Decimal("0")
        self._equity_now = Decimal("0")
        self._realized_pnl = Decimal("0")
        self._last_reset_date = None  # YYYY-MM-DD

    def _maybe_reset(self, now: Optional[datetime] = None) -> None:
        now = now or _now_utc()
        reset_date = now.date()
        if self._last_reset_date != reset_date and now.hour >= self.cfg.reset_h and now.minute >= self.cfg.reset_m:
            # відкриваємо новий день
            self._equity_open = self._equity_now  # фік� уємо � тарт
            self._realized_pnl = Decimal("0")
            self._last_reset_date = reset_date

    def on_portfolio(self, pld: Dict[str, Any]) -> None:
        self._equity_now = _d(pld.get("equity_free_usdt"))
        self._maybe_reset()

    def on_order_filled(self, pld: Dict[str, Any]) -> None:
        # очікуємо pld: {"realized_pnl_usd": "..."} або розрахунок з кумулятивних котирувань
        realized = _d(pld.get("realized_pnl_usd"))
        self._realized_pnl += realized

    def can_open(self) -> Tuple[bool, Dict[str, Any]]:
        # Fail-closed: без коректних equity — блокуємо
        if self._equity_open <= 0 or self._equity_now <= 0:
            return False, {
                "reason": "DAILY_RISK_LIMIT",
                "detail": "NO_EQUITY",
                "equity_open_usd": str(self._equity_open),
                "equity_now_usd": str(self._equity_now),
            }
        dd = Decimal("0")
        if self._equity_open > 0:
            dd = (Decimal("1") - (self._equity_now / self._equity_open)) * Decimal("100")

        if (-self._realized_pnl) >= self.cfg.max_realized_loss_usd:
            return False, {
                "reason": "DAILY_RISK_LIMIT",
                "detail": "MAX_REALIZED_LOSS",
                "realized_pnl_usd": str(self._realized_pnl),
                "limit_usd": str(self.cfg.max_realized_loss_usd),
            }

        if dd >= self.cfg.max_drawdown_pct:
            return False, {
                "reason": "DAILY_RISK_LIMIT",
                "detail": "MAX_DRAWDOWN",
                "drawdown_pct": str(dd),
                "limit_pct": str(self.cfg.max_drawdown_pct),
                "equity_open_usd": str(self._equity_open),
                "equity_now_usd": str(self._equity_now),
            }

        return True, {
            "equity_open_usd": str(self._equity_open),
            "equity_now_usd": str(self._equity_now),
            "realized_pnl_usd": str(self._realized_pnl),
            "drawdown_pct": str(dd),
        }
```

```python
from datetime import datetime, timezone

def _utc_hm() -> tuple[int,int]:
    now = datetime.now(timezone.utc)
    return now.hour, now.minute

def _in_quiet(quiet: list[str]) -> bool:
    h, m = _utc_hm()
    cur = h*60 + m
    for win in quiet or []:
        a,b = win.split("-")
        ah,am = map(int, a.split(":"))
        bh,bm = map(int, b.split(":"))
        start = ah*60+am
        end = bh*60+bm
        if start <= end:
            # звичайний діапазон
            if start <= cur <= end: return True
        else:
            # через північ
            if cur >= start or cur <= end: return True
    return False

# ... у handle() при CMD:OPEN:

            src="execution_position", dst=msg.src, rid=msg.rid,

    # 2) QUIET HOURS

            op="ERR", verb="OPEN", intent="REJECTION",
            src="execution_position", dst=msg.src, rid=msg.rid,
            pld={"reason":"QUIET_HOURS"}, why="quiet_window"

        return
    # 3) ALLOWLIST
    allow = ops_cfg.get("allowlist_symbols") or []
    sym = (msg.pld or {}).get("symbol", "").upper()
    if allow and sym not in set(x.upper() for x in allow):
```

```python
#!/usr/bin/env python3
import os, re, json, sys, time
from urllib.request import urlopen, Request
from urllib.error import URLError

def _get_cfg():
    # про� те читання env з дефолтами; можна замінити на ваш ConfigLoader
    url = os.environ.get("OPS_METRICS_URL", "http://127.0.0.1:8000/metrics")
    out_dir = os.environ.get("OPS_REPORTS_DIR", "reports")
    return url, out_dir

def _scrape_metrics_text(url: str) -> str:
    try:
        req = Request(url, headers={"User-Agent": "metrics-summary/1.0"})
        with urlopen(req, timeout=3) as r:
            return r.read().decode("utf-8", "replace")
    except URLError as e:
        return ""

def _mget(txt: str, name: str, label_filter: str | None = None, default: float = 0.0) -> float:
    """
    Витягає значення метрики з Prometheus-тек� ту.
    Напр., name='exposure_equity_usd'
          name='order_state_total' + label_filter='status="FILLED"'
    """
    if not txt: return default
    if label_filter:
        pat = rf'^{re.escape(name)}\{{[^}}]*{label_filter}[^}}]*\}}\s+([0-9eE\.\+\-]+)$'
    else:
        pat = rf'^{re.escape(name)}\s+([0-9eE\.\+\-]+)$'
    for line in txt.splitlines():
        m = re.match(pat, line)
        if m:
            try: return float(m.group(1))
            except Exception: return default
    return default

def main():
    url, out_dir = _get_cfg()
    txt = _scrape_metrics_text(url)
    # exposure
    equity   = _mget(txt, "exposure_equity_usd")
    pos_usd  = _mget(txt, "exposure_positions_usd")
    pend_usd = _mget(txt, "exposure_pending_usd")
    limit_usd= _mget(txt, "exposure_limit_usd")
    # guards
    exp_rej  = _mget(txt, "fsm_guard_rejects_total", 'guard="exposure"')
    ttl_exp  = _mget(txt, "pending_exposure_expired_total")
    daily_blk= _mget(txt, "fsm_guard_rejects_total", 'guard="daily"')  # якщо дода� те лічильник далі
    # orders
    placed   = _mget(txt, "orders_placed_total")
    filled   = _mget(txt, "orders_filled_total")
    # decisions
    # приклад � имволу — якщо ек� портуєте лейбли по � имволу (необов'язково)
    # rl_eth   = _mget(txt, "decision_rate_limited_total", 'symbol="ETHUSDT"')

    summary = {
        "ts_ms": int(time.time()*1000),
        "exposure": {
            "equity_usd": equity,
            "positions_usd": pos_usd,
            "pending_usd": pend_usd,
            "limit_usd": limit_usd,
            "utilization_pct": ( (pos_usd + pend_usd)/limit_usd*100.0 ) if limit_usd>0 else 0.0
        },
        "guards": {
            "exposure_rejects_total": exp_rej,
            "pending_expired_total": ttl_exp,
            "daily_rejects_total": daily_blk
        },
        "orders": {
            "placed_total": placed,
            "filled_total": filled
        }
    }
    os.makedirs(out_dir, exist_ok=True)
    out_path = os.path.join(out_dir, "summary_gate_status.json")
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, ensure_ascii=False, indent=2)
    print(out_path)

if __name__ == "__main__":
    sys.exit(main())
```
