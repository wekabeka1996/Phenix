    from abc import ABC, abstractmethod
    from vfoundation.core.protocol import Message

    class AbstractExecutionAdapter(ABC):
        """
        Abstract interface for an execution adapter. It defines the contract
        for how the Execution FSM interacts with a trading venue (real or simulated).
        """

        @abstractmethod
        def place_order(self, dec_msg: Message) -> dict:
            """
            Places an order on the exchange based on a DECISION message.

            Args:
                dec_msg: The DEC:OPEN or DEC:ADJUST message from the FSM.

            Returns:
                A dictionary with order confirmation details, e.g.,
                {'status': 'ACCEPTED', 'exchange_order_id': '12345'}.
            """
            pass

        @abstractmethod
        def cancel_order(self, dec_msg: Message) -> dict:
            """
            Cancels an existing order on the exchange.

            Args:
                dec_msg: The DEC:CANCEL message from the FSM.

            Returns:
                A dictionary with cancellation confirmation.
            """
            pass

        @abstractmethod
        def get_status(self) -> str:
            """
            Returns the current status of the adapter connection.
            e.g., 'CONNECTED', 'DISCONNECTED', 'ERROR'.
            """
            pass

    logging.basicConfig(level=logging.INFO)
    logger = logging.getLogger(__name__)

    def find_latest_snapshot(snapshot_dir: str = "ops/snapshots") -> Path | None:
        """Finds the most recent snapshot file in the directory."""
        path = Path(snapshot_dir)
        if not path.exists():
            return None
        
        files = list(path.glob("*.json"))
        if not files:
            return None
            
        return max(files, key=lambda f: f.stat().st_mtime)

    def replay_wal_after(wal_dir: str, start_timestamp_utc: str, target_fsm):
        """Replays WAL entries created after a given timestamp."""
        wal_path = Path(wal_dir)
        start_dt = datetime.fromisoformat(start_timestamp_utc)
        
        files_to_replay = sorted([
            f for f in wal_path.glob("*.jsonl")
            if datetime.fromisoformat(f.stem.replace('_', 'T') + "+00:00") >= start_dt.date()
        ])

        replayed_count = 0
        for file_path in files_to_replay:
            with open(file_path, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        msg_dict = json.loads(line)
                        msg_ts = datetime.fromtimestamp(msg_dict.get("ts", 0) / 1e6, tz=timezone.utc)
                        if msg_ts > start_dt:
                            #                                                                  ,                               
                            if msg_dict.get("name") in ["TRADE_EXECUTED", "ACCOUNT_UPDATE_RECEIVED"]:
                                msg = Message.from_dict(msg_dict)
                                target_fsm.handle_event(msg)
                                replayed_count += 1
                    except (json.JSONDecodeError, KeyError):
                        continue # Ignore corrupted lines
        
        logger.info(f"Replayed {replayed_count} events from WAL.")

    from decimal import Decimal
    from vfoundation.core.fsm import FSM as FSMCore
    from vfoundation.core.protocol import Message

    class RegimeDetector:
        """
        Analyzes market features to detect the current trading regime based on
        pre-configured models.
        """
        def __init__(self, config: dict, fsm: FSMCore):
            """Initializes the detector with model configurations."""
            self.fsm = fsm
            self.config = config
            self.logger = fsm.logger
            self.model_config = self.config.get("models", {}).get("sma_trend", {})
            self.model_name = "sma_trend_v1"

        def _calculate_confidence(self, sma_short: Decimal, sma_long: Decimal) -> Decimal:
            """

            Calculates a confidence score based on the divergence of short and long SMAs.
            The formula is heuristically tuned.
            """
            if sma_long == 0:
                return Decimal("0.5")
            
            # Heuristic formula for confidence based on SMA divergence
            confidence = (sma_short - sma_long) / sma_long * Decimal("20.0")
            
            # Bound the confidence between a minimum floor and a maximum ceiling
            bounded_confidence = min(max(abs(confidence), Decimal("0.5")), Decimal("0.95"))
            return bounded_confidence

        def handle_event(self, event: Message):
            """
            Entry point for handling events. Currently processes FEATURES_CALCULATED.
            """
            if event.name != "FEATURES_CALCULATED":
                return

            features = event.pld.get("features", {})
            price = Decimal(features.get("price", "0"))
            sma_short = Decimal(features.get("sma_short", "0"))
            sma_long = Decimal(features.get("sma_long", "0"))
            symbol = event.pld.get("symbol")
            ts = event.pld.get("ts")

            if not all([price > 0, sma_short > 0, sma_long > 0, symbol, ts]):
                self.logger.debug("Skipping regime detection due to missing feature data.")
                return

            regime = "UNCERTAIN"
            confidence = Decimal("0.5")

            # --- Trend Detection Logic ---
            if sma_short > sma_long and price > sma_short:
                regime = "TREND_UP"

                "regime": regime,

            }
            
            self.fsm.emit("EVT:REGIME_DETECTED", payload, why=f"Regime '{regime}' detected by {self.model_name}")

    def handle_event(self, event: Message):
        """
        Entry point for handling events. Currently processes FEATURES_CALCULATED.
        """
        if event.name != "FEATURES_CALCULATED":
            return

        features = event.pld.get("features", {})
        price = Decimal(features.get("price", "0"))
        sma_short = Decimal(features.get("sma_short", "0"))
        sma_long = Decimal(features.get("sma_long", "0"))
        atr_14 = Decimal(features.get("atr_14", "0"))
        atr_14_sma_100 = Decimal(features.get("atr_14_sma_100", "0"))
        symbol = event.pld.get("symbol")
        ts = event.pld.get("ts")

        if not all([price > 0, symbol, ts]):
            self.logger.debug("Skipping regime detection due to missing core feature data.")
            return

        regime = "UNCERTAIN"
        confidence = Decimal("0.5")
        source_model = self.trend_model_name

        # --- PRIORITY 1: Volatility Detection ---
        if self.vol_model_config.get("enabled") and atr_14 > 0 and atr_14_sma_100 > 0:
            threshold_multiplier = Decimal(self.vol_model_config.get("threshold_multiplier", "2.0"))
            low_vol_multiplier = Decimal(self.vol_model_config.get("low_vol_multiplier", "0.5"))
            volatility_ratio = atr_14 / atr_14_sma_100

            if volatility_ratio > threshold_multiplier:
                regime = "HIGH_VOLATILITY"
                excess_volatility = volatility_ratio - threshold_multiplier
                confidence = min(Decimal("0.95"), Decimal("0.5") + excess_volatility * Decimal("2.0"))
                source_model = self.vol_model_name
            elif volatility_ratio < low_vol_multiplier:
                regime = "LOW_VOLATILITY"
                calm_factor = low_vol_multiplier - volatility_ratio
                confidence = min(Decimal("0.95"), Decimal("0.5") + calm_factor * Decimal("2.0"))
                source_model = self.vol_model_name

        # --- PRIORITY 2 & 3: Mean Reversion & Trend (only if volatility is not decisive) ---
        if regime == "UNCERTAIN" and all([sma_short > 0, sma_long > 0]):
            mean_reversion_threshold = Decimal(self.trend_model_config.get("mean_reversion_threshold", "0.005"))
            sma_spread = abs(sma_short - sma_long) / sma_long
            price_deviation_short = abs(price - sma_short) / sma_short
            price_deviation_long = abs(price - sma_long) / sma_long

            if sma_spread < mean_reversion_threshold and price_deviation_short < mean_reversion_threshold and price_deviation_long < mean_reversion_threshold:
                regime = "MEAN_REVERSION"
                tightness = mean_reversion_threshold - max(sma_spread, price_deviation_short, price_deviation_long)
                confidence = min(Decimal("0.95"), Decimal("0.5") + tightness * Decimal("100.0"))
            elif sma_short > sma_long and price > sma_short:
                regime = "TREND_UP"
                confidence = self._calculate_confidence(sma_short, sma_long)
            elif sma_short < sma_long and price < sma_short:
                regime = "TREND_DOWN"
                confidence = self._calculate_confidence(sma_short, sma_long)

        # --- Emit Event ---
        payload = {
            "ts": ts,
            "symbol": symbol,
            "regime": regime,
            "confidence": str(confidence),
            "source_model": source_model
        }
        
        self.fsm.emit("EVT:REGIME_DETECTED", payload, why=f"Regime '{regime}' detected by {source_model}")
        self.logger.info(f"Emitted EVT:REGIME_DETECTED with regime '{regime}' for {symbol}")

    # PATH: apps/reference/domains/execution_position/execution_adapter.py
    from abc import ABC, abstractmethod
    from vfoundation.core.protocol import Message
    from typing import Dict, Any #              typing        AdapterResponse

    #                                                                         
    AdapterResponse = Dict[str, Any]

    class AbstractExecutionAdapter(ABC):
        """
        Abstract interface for an execution adapter. It defines the contract
        for how the Execution FSM interacts with a trading venue (real or simulated).
        """

        @abstractmethod
        def place_order(self, dec_msg: Message) -> AdapterResponse:
            """
            Places an order on the exchange based on a DECISION message.

            Args:
                dec_msg: The DEC:OPEN or DEC:ADJUST message from the FSM.

            Returns:
                A dictionary with order confirmation details from the exchange, e.g.,
                {'status': 'ACCEPTED', 'exchange_order_id': '12345',
                 'filled_qty': '0.0', 'message': '...', 'timestamp': 123456789}.
                 Status can be 'ACCEPTED', 'REJECTED', 'ERROR'.
                 filled_qty should be a string representation of a Decimal.
            """
            pass

        @abstractmethod
        def cancel_order(self, dec_msg: Message) -> AdapterResponse:
            """
            Cancels an existing order on the exchange.

            Args:
                dec_msg: The DEC:CANCEL message from the FSM, containing
                         at least {'exchange_order_id': '...'}.

            Returns:
                A dictionary with cancellation confirmation, similar structure
                to place_order response. Status indicates acceptance of cancel request.
            """
            pass

        @abstractmethod
        def get_status(self) -> str:
            """
            Returns the current status of the adapter connection.
            e.g., 'CONNECTED', 'DISCONNECTED', 'CONNECTING', 'ERROR'.
            """
            pass
